import Lean.Util.CollectAxioms
import Lean.Elab.Command

namespace SharedUnionAuditSmoke

def CleanStatement : Prop :=
  (True = True) ∧ (∃ _ : Unit, True) ∧
    (Quot.mk (fun _ _ : Unit => True) () = Quot.mk (fun _ _ : Unit => True) ())

private theorem cleanShared : CleanStatement :=
  ⟨propext Iff.rfl,
    ⟨Classical.choice (show Nonempty Unit from ⟨()⟩), True.intro⟩,
    @Quot.sound Unit (fun _ _ => True) () () True.intro⟩

opaque cleanOpaque : CleanStatement := cleanShared

theorem cleanA : CleanStatement := cleanOpaque
theorem cleanB : CleanStatement := cleanOpaque

axiom forbiddenProof : True
private theorem hiddenShared : CleanStatement ∧ True := ⟨cleanOpaque, forbiddenProof⟩
opaque hiddenOpaque : CleanStatement ∧ True := hiddenShared

theorem hiddenA : CleanStatement ∧ True := hiddenOpaque
theorem hiddenB : CleanStatement ∧ True := hiddenOpaque

axiom forbiddenType : Prop
-- The axiom appears in the declared type while the proof body is just cleanOpaque.
theorem typeA : (fun _ : Prop => CleanStatement) forbiddenType := cleanOpaque
theorem typeB : (fun _ : Prop => CleanStatement) forbiddenType := cleanOpaque

open Lean Elab Command in
run_cmd do
  let allowed := #[``propext, ``Classical.choice, ``Quot.sound]
  let targets := #[``typeA, ``typeB]
  let env ← getEnv
  let (_, axiomState) := ((targets.forM Lean.CollectAxioms.collect).run env).run {}
  for n in allowed do
    unless axiomState.axioms.contains n do
      throwError "Fixture failed to exercise allowed axiom {n}"
  unless axiomState.visited.contains ``cleanOpaque && axiomState.visited.contains ``cleanShared do
    throwError "Shared private/opaque clean dependencies were not traversed"
  for target in targets do
    let ci ← getConstInfo target
    unless ci.type.getUsedConstants.contains ``forbiddenType do
      throwError "Type-only fixture lost the axiom in the declared type"
    let some value := ci.value? true | throwError "Missing fixture proof body"
    if value.getUsedConstants.contains ``forbiddenType then
      throwError "Type-only fixture unexpectedly uses its test axiom directly in its body"
  unless axiomState.axioms.contains ``forbiddenType do
    throwError "Fixture test axiom was not collected"
  unless axiomState.axioms.size == 4 do
    throwError "Unexpected fixture axiom count {axiomState.axioms}"
  logInfo m!"SHARED_UNION_SMOKE: targets={targets}; collected={axiomState.axioms}"
  let forbidden := axiomState.axioms.filter fun n => !(#[``propext, ``Classical.choice, ``Quot.sound]).contains n
  unless forbidden.isEmpty do
    throwError "Unexpected axiom in endpoint union: {forbidden}"

end SharedUnionAuditSmoke
